const fs=require('fs'),assert=require('assert/strict'),root='/home/tetsuo/grok/thursday-arena',dir=root+'/docs/results/live-fixes-2026-09-24';
const {clusteredInterval}=require(root+'/tools/eval_versions');const season=Number(process.argv[2]||4),seeds=season===4?[211,223,227,229,233]:[241,251,257];const csv=['season,seed,match,handle,before,current,before_score,current_score,delta'];
const counts={before:{win:0,draw:0,loss:0},current:{win:0,draw:0,loss:0}},deltas=[],seedMeans=[],pairs=[],diag={};const score=x=>x==='win'?1:x==='draw'?.5:0;
for(const seed of seeds){const arms=Object.fromEntries(['before','current'].map(name=>[name,JSON.parse(fs.readFileSync(`${dir}/s${season}-${seed}-${name}.json`))]));const a=arms.before,b=arms.current;
 assert.equal(a.stats.matches.length,20);assert.equal(b.stats.matches.length,20);assert.deepEqual(a.starts,b.starts,'external draws must match');let sum=0;
 for(let i=0;i<20;i++){const x=a.stats.matches[i],y=b.stats.matches[i];assert.equal(x.handle,y.handle);const d=score(y.result)-score(x.result);counts.before[x.result]++;counts.current[y.result]++;sum+=d;deltas.push(d);pairs.push({seed,match:i+1,before:score(x.result),current:score(y.result),delta:d});csv.push([season,seed,i+1,x.handle,x.result,y.result,score(x.result),score(y.result),d].join(','));}
 seedMeans.push(sum/20);
 for(const [name,arm] of Object.entries(arms)){const row=diag[name]||={illegal:0,fillableShort:0,missingInputs:0,emptyGhosts:0,rounds:0};row.illegal+=arm.stats.illegal;row.fillableShort+=arm.stats.endShops.filter(s=>s.units<3&&s.cheapestOffer!=null&&Number.isFinite(s.cheapestOffer)&&s.goldLeft>=s.cheapestOffer).length;row.missingInputs+=arm.missingInputs;row.emptyGhosts+=arm.emptyGhosts;row.rounds+=arm.stats.rounds.length;}
}
const summary={n:deltas.length,counts,matchPoints:Object.fromEntries(Object.entries(counts).map(([k,v])=>[k,v.win+.5*v.draw])),scoreDelta:deltas.reduce((a,b)=>a+b,0)/deltas.length,seedMeans,seedClustered95:clusteredInterval(seedMeans),better:deltas.filter(d=>d>0).length,worse:deltas.filter(d=>d<0).length,same:deltas.filter(d=>d===0).length,diagnostics:diag,halves:[0,1].map(h=>({half:h+1,...Object.fromEntries(['before','current'].map(k=>[k,pairs.filter(r=>h===0?r.match<=10:r.match>10).reduce((n,r)=>n+r[k],0)/(seeds.length*10)]))}))};
fs.writeFileSync(dir+`/paired-s${season}.csv`,csv.join('\n')+'\n');fs.writeFileSync(dir+`/paired-s${season}-summary.json`,JSON.stringify(summary,null,2)+'\n');fs.writeFileSync(dir+`/paired-s${season}.log`,JSON.stringify(summary,null,2)+'\n');console.log(JSON.stringify(summary,null,2));
